   170→   170→function allowedNextStatuses(current: string): string[] {
   171→   171→  if (current === 'CANCELLED') return []
   172→   172→  const idx = STATUS_FLOW.indexOf(current as (typeof STATUS_FLOW)[number])
   173→   173→  const forward = idx >= 0 ? STATUS_FLOW.slice(idx + 1) : STATUS_FLOW
   174→   174→  return [...forward, 'CANCELLED']
   175→   175→}
