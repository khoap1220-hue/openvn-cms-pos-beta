   165→   165→      setLoading(false)
   166→   166→    }
   167→   167→  }, [year, quarter, search])

   169→   169→  useEffect(() => {
   170→   170→    void load()

   181→   181→      setProducts(productsRes.products)

   190→   190→    setDialogOpen(true)

   192→   192→  }

   194→   194→  const updateRow = (index: number, patch: Partial<DraftRow>) =>
   195→   195→    setRows((prev) => prev.map((row, i) => (i === index ? { ...row, ...patch } : row)))
   196→   196→

   204→   204→      (products ?? []).find((p) => p.barcode === code) ?? (products ?? []).find((p) => p.sku === code)

   206→   206→      toast({ title: `Không tìm thấy sản phẩm với mã vạch ${code}`, variant: 'destructive' })

   212→   212→        // Đã có dòng của SP này → tăng số lượng lên 1

   214→   214→          i === existIdx ? { ...row, quantity: String((Number(row.quantity) || 0) + 1) } : row

   216→   216→      }

   219→   219→        return prev.map((row, i) =>
   220→   220→          i === 0 ? { productId: product.id, quantity: '1', unitPrice: String(product.costPrice) } : row

   222→   222→      }
   223→   223→      return [...prev, { productId: product.id, quantity: '1', unitPrice: String(product.costPrice) }]
   224→   224→    })

   245→   245→
   246→   246→  const handleCreate = async () => {

   249→   249→      const qty = Number(row.quantity)
   250→   250→      const price = row.unitPrice === '' ? undefined : Number(row.unitPrice)

   252→   252→        toast({ title: 'Vui lòng chọn sản phẩm cho mọi dòng', variant: 'destructive' })

   254→   254→      }

   256→   256→        toast({ title: 'Số lượng phải lớn hơn 0', variant: 'destructive' })

   260→   260→        toast({ title: 'Đơn giá không được âm', variant: 'destructive' })
   261→   261→        return
   262→   262→      }
   263→   263→      items.push({ productId: row.productId, quantity: qty, ...(price !== undefined ? { unitPrice: price } : {}) })
   264→   264→    }
   265→   265→    if (items.length === 0) {
   266→   266→      toast({ title: 'Cần ít nhất 1 dòng sản phẩm', variant: 'destructive' })

   273→   273→        items,
   274→   274→        vatRate: Number(form.vatRate) || 0,
   275→   275→        invoiceDate: form.invoiceDate || undefined,
   276→   276→        note: form.note.trim() || undefined,

   279→   279→        title: 'Đã ghi sổ hóa đơn mua',
   280→   280→        description: 'Chi phí & tồn kho đã được cập nhật theo hóa đơn.',

   283→   283→      await load()

   285→   285→      toast({
   286→   286→        title: 'Ghi sổ hóa đơn mua thất bại',
   287→   287→        description: e instanceof Error ? e.message : 'Vui lòng thử lại',
   288→   288→        variant: 'destructive',
   289→   289→      })
   290→   290→    } finally {
   291→   291→      setSubmitting(false)
   292→   292→    }
   293→   293→  }

   301→   301→        title: 'Đã hủy hóa đơn mua',

   305→   305→      setDetail(null)
   306→   306→      await load()
   307→   307→    } catch (e) {
   308→   308→      toast({
   309→   309→        title: 'Hủy hóa đơn thất bại',
   310→   310→        description: e instanceof Error ? e.message : 'Vui lòng thử lại',
   311→   311→        variant: 'destructive',
   312→   312→      })
   313→   313→    } finally {
   314→   314→      setCancelling(false)
   315→   315→    }
   316→   316→  }
   317→   317→
   318→   318→  const selectProduct = (index: number, productId: string) => {

   320→   320→    updateRow(index, {
   321→   321→      productId,
   322→   322→      unitPrice:
   323→   323→        product && (rows[index].unitPrice === '' || Number(rows[index].unitPrice) === 0)
   324→   324→          ? String(product.costPrice)
   325→   325→          : rows[index].unitPrice,
   326→   326→    })
   327→   327→  }

   359→   359→          value={loading ? '' : formatCompactVND(totalVatInput)}

   373→   373→
   374→   374→      {/* Bộ lọc */}
   375→   375→      <div className="mb-4 flex flex-wrap items-center gap-3">

   380→   380→          <SelectContent>

   383→   383→                {q.label}
   384→   384→              </SelectItem>
   385→   385→            ))}

   387→   387→        </Select>

   396→   396→              </SelectItem>

   398→   398→          </SelectContent>
   399→   399→        </Select>
   400→   400→        <div className="relative w-full max-w-xs">

   411→   411→      <Card className="border-border/70 shadow-sm">
   412→   412→        <CardContent className="p-0">

   416→   416→            </div>
   417→   417→          ) : error ? (

   419→   419→              <ErrorState message={error} onRetry={() => void load()} />
   420→   420→            </div>
   421→   421→          ) : !invoices || invoices.length === 0 ? (
   422→   422→            <div className="p-4">
   423→   423→              <EmptyState
   424→   424→                icon={PackagePlus}

   427→   427→                action={
   428→   428→                  <Button onClick={openCreate}>
   429→   429→                    <PackagePlus className="h-4 w-4" />
   430→   430→                    Tạo hóa đơn mua

   439→   439→                  <TableRow>

   441→   441→                    <TableHead>Ngày</TableHead>
   442→   442→                    <TableHead>Đối tác</TableHead>
