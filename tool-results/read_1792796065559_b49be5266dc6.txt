    68→  code: string

    70→  taxCode: string | null

    73→
    74→interface InvoiceItemRow {
    75→  quantity: number
    76→  unitPrice: number
    77→  amount: number
    78→  product: { sku: string; name: string; unit: string }

    81→interface SalesInvoice {
    82→  id: string
    83→  code: string
    84→  invoiceDate: string
    85→  subtotal: number
    86→  vatRate: number
    87→  vatAmount: number
    88→  total: number
    89→  status: 'POSTED' | 'CANCELLED'

    99→  subtotal: number

   102→  createdAt: string

   104→
   105→interface DraftRow {
   106→  productId: string
   107→  quantity: string
   108→  unitPrice: string
   109→}

   111→const FIRST_YEAR = 2024
   112→const CURRENT_YEAR = new Date().getFullYear()
   113→const YEAR_OPTIONS = Array.from({ length: CURRENT_YEAR + 2 - FIRST_YEAR }, (_, i) => FIRST_YEAR + i)
   114→const QUARTER_OPTIONS = [
   115→  { value: 'all', label: 'Cả năm' },
   116→  { value: '1', label: 'Quý 1' },
   117→  { value: '2', label: 'Quý 2' },
   118→  { value: '3', label: 'Quý 3' },
   119→  { value: '4', label: 'Quý 4' },
   120→]
   121→const VAT_RATES = [0, 5, 8, 10]
   122→
   123→const todayInput = () => {
   124→  const d = new Date()
   125→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   126→  const dd = String(d.getDate()).padStart(2, '0')
   127→  return `${d.getFullYear()}-${mm}-${dd}`
   128→}
   129→
   130→export function SalesView() {
   131→  const [invoices, setInvoices] = useState<SalesInvoice[] | null>(null)
   132→  const [uninvoiced, setUninvoiced] = useState<UninvoicedOrder[] | null>(null)
   133→  const [loading, setLoading] = useState(true)
   134→  const [error, setError] = useState<string | null>(null)
   135→
   136→  const [year, setYear] = useState<number>(CURRENT_YEAR)
   137→  const [quarter, setQuarter] = useState<string>('all')
   138→  const [searchInput, setSearchInput] = useState('')
   139→  const [search, setSearch] = useState('')
   140→
   141→  const [partners, setPartners] = useState<PartnerLite[] | null>(null)
   142→  const [products, setProducts] = useState<ProductLite[] | null>(null)
   143→
   144→  const [detail, setDetail] = useState<SalesInvoice | null>(null)
   145→  const [cancelTarget, setCancelTarget] = useState<SalesInvoice | null>(null)
   146→  const [cancelling, setCancelling] = useState(false)
   147→
   148→  const [dialogOpen, setDialogOpen] = useState(false)
   149→  const [dialogTab, setDialogTab] = useState<'order' | 'manual'>('order')
   150→  const [selectedOrderId, setSelectedOrderId] = useState<string>('')
   151→  const [manualPartnerId, setManualPartnerId] = useState('none')
   152→  const [invoiceDate, setInvoiceDate] = useState(todayInput())
   153→  const [vatRate, setVatRate] = useState('8')
   154→  const [note, setNote] = useState('')
   155→  const [rows, setRows] = useState<DraftRow[]>([{ productId: '', quantity: '', unitPrice: '' }])
   156→  const [submitting, setSubmitting] = useState(false)
   157→  const { toast } = useToast()
   158→
   159→  useEffect(() => {
   160→    const t = setTimeout(() => setSearch(searchInput.trim()), 400)
   161→    return () => clearTimeout(t)
   162→  }, [searchInput])
   163→
   164→  const load = useCallback(async () => {
   165→    setLoading(true)
   166→    setError(null)
   167→    try {
   168→      const params = new URLSearchParams({ year: String(year) })
   169→      if (quarter !== 'all') params.set('quarter', quarter)
   170→      if (search) params.set('q', search)
   171→      const [invoiceRes, uninvoicedRes] = await Promise.all([
   172→        apiGet<{ invoices: SalesInvoice[] }>(`/api/sales-invoices?${params.toString()}`),
   173→        apiGet<{ orders: UninvoicedOrder[] }>('/api/sales-invoices?uninvoiced=1'),
   174→      ])
   175→      setInvoices(invoiceRes.invoices)
   176→      setUninvoiced(uninvoicedRes.orders)
   177→    } catch (e) {
   178→      const message = e instanceof Error ? e.message : 'Không thể tải hóa đơn bán'
   179→      setError(message)
   180→    } finally {
   181→      setLoading(false)
   182→    }
   183→  }, [year, quarter, search])
   184→
   185→  useEffect(() => {
   186→    void load()
   187→  }, [load])
   188→
   189→  const loadDialogData = useCallback(async () => {
   190→    try {
   191→      const [partnersRes, productsRes] = await Promise.all([
   192→        apiGet<{ partners: PartnerLite[] }>('/api/partners'),
   193→        apiGet<{ products: ProductLite[] }>('/api/products?status=ACTIVE'),
   194→      ])
   195→      setPartners(partnersRes.partners.filter((p) => p.type === 'CUSTOMER' || p.type === 'BOTH'))
   196→      setProducts(productsRes.products)
   197→    } catch {
   198→      // bỏ qua — dialog vẫn mở, select sẽ rỗng
   199→    }
   200→  }, [])
   201→
   202→  const openCreate = () => {
   203→    setDialogTab('order')
   204→    setSelectedOrderId('')
   205→    setManualPartnerId('none')
   206→    setInvoiceDate(todayInput())
   207→    setVatRate('8')
   208→    setNote('')
   209→    setRows([{ productId: '', quantity: '', unitPrice: '' }])
   210→    setDialogOpen(true)
   211→    void loadDialogData()
   212→  }
   213→
   214→  const updateRow = (index: number, patch: Partial<DraftRow>) =>
   215→    setRows((prev) => prev.map((row, i) => (i === index ? { ...row, ...patch } : row)))
   216→
   217→  const removeRow = (index: number) => setRows((prev) => prev.filter((_, i) => i !== index))
   218→
   219→  const addRow = () => setRows((prev) => [...prev, { productId: '', quantity: '', unitPrice: '' }])
   220→
   221→  const selectProduct = (index: number, productId: string) => {
   222→    const product = (products ?? []).find((p) => p.id === productId)
   223→    updateRow(index, {
   224→      productId,
   225→      unitPrice:
   226→        product && (rows[index].unitPrice === '' || Number(rows[index].unitPrice) === 0)
   227→          ? String(product.sellPrice)
   228→          : rows[index].unitPrice,
   229→    })
   230→  }
   231→
   232→  const manualTotals = useMemo(() => {
   233→    let subtotal = 0
   234→    for (const row of rows) {
   235→      subtotal += (Number(row.quantity) || 0) * (Number(row.unitPrice) || 0)
   236→    }
   237→    const vat = Math.round((subtotal * (Number(vatRate) || 0)) / 100)
   238→    return { subtotal, vat, total: subtotal + vat }
   239→  }, [rows, vatRate])
   240→
   241→  const posted = (invoices ?? []).filter((inv) => inv.status === 'POSTED')
   242→  const totalRevenue = posted.reduce((sum, inv) => sum + inv.total, 0)
   243→  const totalVatOutput = posted.reduce((sum, inv) => sum + inv.vatAmount, 0)
   244→
   245→  const handleCreate = async () => {
   246→    const common = {
   247→      vatRate: Number(vatRate) || 0,
   248→      invoiceDate: invoiceDate || undefined,
   249→      note: note.trim() || undefined,
   250→    }
   251→    let body: Record<string, unknown>
   252→    if (dialogTab === 'order') {
   253→      if (!selectedOrderId) {
   254→        toast({ title: 'Vui lòng chọn một đơn hàng cần xuất hóa đơn', variant: 'destructive' })
   255→        return
   256→      }
   257→      body = { orderId: selectedOrderId, ...common }
   258→    } else {
   259→      const items: Array<{ productId: string; quantity: number; unitPrice?: number }> = []
   260→      for (const row of rows) {
   261→        const qty = Number(row.quantity)
   262→        const price = row.unitPrice === '' ? undefined : Number(row.unitPrice)
   263→        if (!row.productId) {
   264→          toast({ title: 'Vui lòng chọn sản phẩm cho mọi dòng', variant: 'destructive' })
   265→          return
   266→        }
   267→        if (!Number.isFinite(qty) || qty <= 0) {
   268→          toast({ title: 'Số lượng phải lớn hơn 0', variant: 'destructive' })
   269→          return
   270→        }
   271→        if (price !== undefined && (!Number.isFinite(price) || price < 0)) {
   272→          toast({ title: 'Đơn giá không được âm', variant: 'destructive' })
   273→          return
   274→        }
   275→        items.push({ productId: row.productId, quantity: qty, ...(price !== undefined ? { unitPrice: price } : {}) })
   276→      }
   277→      if (items.length === 0) {
   278→        toast({ title: 'Cần ít nhất 1 dòng sản phẩm', variant: 'destructive' })
   279→        return
   280→      }
   281→      body = {
   282→        partnerId: manualPartnerId !== 'none' ? manualPartnerId : undefined,
   283→        items,
   284→        ...common,
   285→      }
   286→    }
   287→    setSubmitting(true)
   288→    try {
   289→      await apiPost<{ invoice: SalesInvoice }>('/api/sales-invoices', body)
   290→      toast({
   291→        title: 'Đã xuất hóa đơn bán',
   292→        description: 'Doanh thu & thuế VAT đầu ra đã được ghi nhận.',
   293→      })
   294→      setDialogOpen(false)
   295→      await load()
   296→    } catch (e) {
   297→      toast({
   298→        title: 'Xuất hóa đơn bán thất bại',
   299→        description: e instanceof Error ? e.message : 'Vui lòng thử lại',
   300→        variant: 'destructive',
   301→      })
   302→    } finally {
   303→      setSubmitting(false)
   304→    }
   305→  }
   306→
   307→  const handleCancel = async () => {
   308→    if (!cancelTarget) return
   309→    setCancelling(true)
   310→    try {
   311→      await apiPost<{ invoice: SalesInvoice }>(`/api/sales-invoices/${cancelTarget.id}`, { action: 'CANCEL' })
   312→      toast({
   313→        title: 'Đã hủy hóa đơn bán',
   314→        description: `${cancelTarget.code} — doanh thu, VAT & kho đã hoàn ngược.`,
   315→      })
   316→      setCancelTarget(null)
   317→      setDetail(null)
   318→      await load()
   319→    } catch (e) {
   320→      toast({
   321→        title: 'Hủy hóa đơn thất bại',
   322→        description: e instanceof Error ? e.message : 'Vui lòng thử lại',
   323→        variant: 'destructive',
   324→      })
   325→    } finally {
   326→      setCancelling(false)
   327→    }
   328→  }
   329→
   330→  return (
   331→    <div>
   332→      <PageHeader
   333→        title="Xuất hàng"
   334→        description="Hóa đơn bán — ghi sổ doanh thu, thuế VAT đầu ra và trừ kho (kèm định mức BOM)"
   335→      >
   336→        <Button onClick={openCreate}>
   337→          <Receipt className="h-4 w-4" />
   338→          Tạo hóa đơn bán
   339→        </Button>
   340→      </PageHeader>
   341→
   342→      <div className="mb-6 grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   343→        <StatCard
   344→          title="Tổng thu"
   345→          value={loading ? '' : formatCompactVND(totalRevenue)}
   346→          icon={CircleDollarSign}
   347→          loading={loading}
   348→          hint="Đã loại hóa đơn hủy"
   349→          iconClassName="bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400"
   350→        />
   351→        <StatCard
   352→          title="Số hóa đơn"
   353→          value={loading ? '' : formatNumber(invoices?.length ?? 0)}
   354→          icon={Receipt}
   355→          loading={loading}
   356→          hint={`${posted.length} đã ghi sổ`}
   357→        />
   358→        <StatCard
   359→          title="Thuế VAT đầu ra"
   360→          value={loading ? '' : formatCompactVND(totalVatOutput)}
   361→          icon={FileSpreadsheet}
   362→          loading={loading}
   363→          hint="Kê khai từ hóa đơn bán"
   364→          iconClassName="bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-400"
   365→        />
   366→        <StatCard
   367→          title="Đơn chưa xuất HĐ"
   368→          value={loading ? '' : formatNumber(uninvoiced?.length ?? 0)}
   369→          icon={ClipboardList}
   370→          loading={loading}
   371→          hint="Đơn COMPLETED chưa có hóa đơn"
   372→          iconClassName="bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-400"
   373→        />
   374→      </div>
   375→
   376→      {/* Bộ lọc */}
   377→      <div className="mb-4 flex flex-wrap items-center gap-3">
   378→        <Select value={quarter} onValueChange={setQuarter}>
   379→          <SelectTrigger className="w-[140px]">
   380→            <SelectValue />
   381→          </SelectTrigger>
   382→          <SelectContent>
   383→            {QUARTER_OPTIONS.map((q) => (
   384→              <SelectItem key={q.value} value={q.value}>
   385→                {q.label}
   386→              </SelectItem>
   387→            ))}
   388→          </SelectContent>
   389→        </Select>
   390→        <Select value={String(year)} onValueChange={(value) => setYear(Number(value))}>
   391→          <SelectTrigger className="w-[120px]">
   392→            <SelectValue />
   393→          </SelectTrigger>
   394→          <SelectContent>
   395→            {YEAR_OPTIONS.map((y) => (
   396→              <SelectItem key={y} value={String(y)}>
   397→                Năm {y}
   398→              </SelectItem>
   399→            ))}
   400→          </SelectContent>
   401→        </Select>
   402→        <div className="relative w-full max-w-xs">
   403→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   404→          <Input
   405→            value={searchInput}
   406→            onChange={(e) => setSearchInput(e.target.value)}
   407→            placeholder="Tìm mã hóa đơn, đối tác, đơn nguồn..."
   408→            className="pl-8"
   409→          />
   410→        </div>
   411→      </div>
   412→
   413→      <Card className="border-border/70 shadow-sm">
   414→        <CardContent className="p-0">
   415→          {loading ? (
   416→            <div className="p-4">
   417→              <TableSkeleton rows={6} />
   418→            </div>
   419→          ) : error ? (
   420→            <div className="p-4">
   421→              <ErrorState message={error} onRetry={() => void load()} />
   422→            </div>
   423→          ) : !invoices || invoices.length === 0 ? (
   424→            <div className="p-4">
   425→              <EmptyState
   426→                icon={Receipt}
   427→                title="Chưa có hóa đơn bán nào"
   428→                description="Xuất hóa đơn từ đơn hàng COMPLETED hoặc tự tạo hóa đơn thủ công."
   429→                action={
   430→                  <Button onClick={openCreate}>
   431→                    <Receipt className="h-4 w-4" />
   432→                    Tạo hóa đơn bán
   433→                  </Button>
   434→                }
   435→              />
   436→            </div>
   437→          ) : (
   438→            <div className="max-h-96 overflow-auto">
   439→              <Table>
   440→                <TableHeader className="sticky top-0 bg-background">
   441→                  <TableRow>
   442→                    <TableHead>Mã HĐ</TableHead>
   443→                    <TableHead>Ngày</TableHead>
   444→                    <TableHead>Khách hàng</TableHead>
   445→                    <TableHead>Đơn nguồn</TableHead>
   446→                    <TableHead className="text-right">Tiền hàng</TableHead>
   447→                    <TableHead className="text-right">VAT</TableHead>
   448→                    <TableHead className="text-right">Tổng cộng</TableHead>
   449→                    <TableHead>Trạng thái</TableHead>
   450→                    <TableHead className="w-12" />
   451→                  </TableRow>
   452→                </TableHeader>
   453→                <TableBody>
   454→                  {invoices.map((inv) => (
   455→                    <TableRow key={inv.id} className={inv.status === 'CANCELLED' ? 'opacity-60' : undefined}>
   456→                      <TableCell className="whitespace-nowrap font-mono text-xs font-medium">{inv.code}</TableCell>
   457→                      <TableCell className="whitespace-nowrap text-sm">{formatDate(inv.invoiceDate)}</TableCell>
   458→                      <TableCell>
   459→                        <p className="max-w-[160px] truncate font-medium">{inv.partner?.name ?? 'Khách lẻ'}</p>
   460→                        {inv.partner?.taxCode && (
   461→                          <p className="font-mono text-xs text-muted-foreground">{inv.partner.taxCode}</p>
   462→                        )}
   463→                      </TableCell>
   464→                      <TableCell className="whitespace-nowrap font-mono text-xs">
   465→                        {inv.order?.code ?? '—'}
   466→                      </TableCell>
   467→                      <TableCell className="whitespace-nowrap text-right tabular-nums">{formatVND(inv.subtotal)}</TableCell>
   468→                      <TableCell className="whitespace-nowrap text-right tabular-nums text-sky-700 dark:text-sky-400">
   469→                        {inv.vatRate}% · {formatVND(inv.vatAmount)}
   470→                      </TableCell>
   471→                      <TableCell className="whitespace-nowrap text-right font-semibold tabular-nums">
   472→                        {formatVND(inv.total)}
   473→                      </TableCell>
   474→                      <TableCell>
   475→                        {inv.status === 'POSTED' ? (
   476→                          <Badge className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400">
   477→                            Đã ghi sổ
   478→                          </Badge>
   479→                        ) : (
   480→                          <Badge variant="secondary">Đã hủy</Badge>
   481→                        )}
   482→                      </TableCell>
   483→                      <TableCell>
   484→                        <Button
   485→                          variant="ghost"
   486→                          size="icon"
   487→                          className="h-8 w-8"
   488→                          onClick={() => setDetail(inv)}
   489→                          title="Xem chi tiết"
   490→                        >
   491→                          <Eye className="h-4 w-4" />
   492→                        </Button>
   493→                      </TableCell>
   494→                    </TableRow>
   495→                  ))}
   496→                </TableBody>
   497→              </Table>
   498→            </div>
   499→          )}
   500→        </CardContent>
   501→      </Card>
   502→
   503→      {/* Dialog chi tiết hóa đơn bán */}
   504→      <Dialog open={detail !== null} onOpenChange={(open) => !open && setDetail(null)}>
   505→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-2xl">
   506→          {detail && (
   507→            <>
   508→              <DialogHeader>
   509→                <DialogTitle className="flex flex-wrap items-center gap-2">
   510→                  <span className="font-mono">{detail.code}</span>
   511→                  {detail.status === 'POSTED' ? (
   512→                    <Badge className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400">
   513→                      Đã ghi sổ
   514→                    </Badge>
   515→                  ) : (
   516→                    <Badge variant="secondary">Đã hủy</Badge>
   517→                  )}
   518→                </DialogTitle>
   519→                <DialogDescription>
   520→                  {formatDate(detail.invoiceDate)} — {detail.partner?.name ?? 'Khách lẻ'}
   521→                  {detail.order ? ` · từ đơn ${detail.order.code}` : ' · tạo thủ công'}
   522→                </DialogDescription>
   523→              </DialogHeader>
   524→              <div className="overflow-x-auto">
   525→                <Table>
   526→                  <TableHeader>
   527→                    <TableRow>
   528→                      <TableHead className="font-mono">Mã SP</TableHead>
   529→                      <TableHead>Sản phẩm</TableHead>
