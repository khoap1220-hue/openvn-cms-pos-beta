   182→// Các trạng thái đích hợp lệ từ trạng thái hiện tại (chỉ tiến theo luồng + có thể hủy)
   183→function allowedNextStatuses(current: string): string[] {
   184→  if (current === 'CANCELLED') return []
   185→  const idx = STATUS_FLOW.indexOf(current as (typeof STATUS_FLOW)[number])
   186→  const forward = idx >= 0 ? STATUS_FLOW.slice(idx + 1) : STATUS_FLOW
   187→  return [...forward, 'CANCELLED']
   188→}

   190→function itemsSummary(order: Order): string {
   191→  if (order.items.length === 0) return '—'
   192→  if (order.items.length === 1) {
   193→    const item = order.items[0]
   194→    return `${item.quantity} × ${item.product?.name ?? 'Sản phẩm'}`
   195→  }
   196→  return `${order.items.length} sản phẩm`
   197→}
