   209→// ===== Helpers =====

   211→function errorMessage(err: unknown): string {
   212→  return err instanceof Error ? err.message : 'Đã xảy ra lỗi'
   213→}
   214→
   215→/** Parse chi tiết JSON an toàn để hiển thị pretty */
   216→function prettyDetail(detail: string | null): string {
   217→  if (!detail) return '—'
   218→  try {
   219→    const parsed: unknown = JSON.parse(detail)
   220→    if (parsed !== null && typeof parsed === 'object') return JSON.stringify(parsed, null, 2)
   221→    return String(parsed)
   222→  } catch {
   223→    return detail
   224→  }
   225→}
   226→
   227→function sampleText(value: unknown): string {
   228→  if (value === null || value === undefined) return '—'
   229→  if (typeof value === 'number') return formatNumber(value)
   230→  return String(value)
   231→}
