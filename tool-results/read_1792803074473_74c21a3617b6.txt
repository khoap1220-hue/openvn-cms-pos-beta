   100→const FIRST_YEAR = 2024
   101→const CURRENT_YEAR = new Date().getFullYear()
   102→const YEAR_OPTIONS = Array.from({ length: CURRENT_YEAR + 2 - FIRST_YEAR }, (_, i) => FIRST_YEAR + i)
   103→const QUARTER_OPTIONS = [
   104→  { value: 'all', label: 'Cả năm' },
   105→  { value: '1', label: 'Quý 1' },
   106→  { value: '2', label: 'Quý 2' },
   107→  { value: '3', label: 'Quý 3' },
   108→  { value: '4', label: 'Quý 4' },
   109→]
   110→const VAT_RATES = [0, 5, 8, 10]
   111→
   112→const todayInput = () => {
   113→  const d = new Date()
   114→  const mm = String(d.getMonth() + 1).padStart(2, '0')
   115→  const dd = String(d.getDate()).padStart(2, '0')
   116→  return `${d.getFullYear()}-${mm}-${dd}`
   117→}

   119→export function PurchaseView() {
   120→  const [invoices, setInvoices] = useState<PurchaseInvoice[] | null>(null)
   121→  const [loading, setLoading] = useState(true)
   122→  const [error, setError] = useState<string | null>(null)
   123→
   124→  const [year, setYear] = useState<number>(CURRENT_YEAR)
   125→  const [quarter, setQuarter] = useState<string>('all')
   126→  const [searchInput, setSearchInput] = useState('')
   127→  const [search, setSearch] = useState('')
   128→
   129→  const [partners, setPartners] = useState<PartnerLite[] | null>(null)
   130→  const [products, setProducts] = useState<ProductLite[] | null>(null)
   131→
   132→  const [detail, setDetail] = useState<PurchaseInvoice | null>(null)
   133→  const [cancelTarget, setCancelTarget] = useState<PurchaseInvoice | null>(null)
   134→  const [cancelling, setCancelling] = useState(false)
   135→
   136→  const [dialogOpen, setDialogOpen] = useState(false)
   137→  const [form, setForm] = useState<{
   138→    partnerId: string
   139→    invoiceDate: string
   140→    vatRate: string
   141→    note: string
   142→  }>({ partnerId: 'none', invoiceDate: todayInput(), vatRate: '8', note: '' })
   143→  const [rows, setRows] = useState<DraftRow[]>([{ productId: '', quantity: '', unitPrice: '' }])
   144→  const [submitting, setSubmitting] = useState(false)
   145→  const { toast } = useToast()
   146→
   147→  useEffect(() => {
   148→    const t = setTimeout(() => setSearch(searchInput.trim()), 400)
   149→    return () => clearTimeout(t)
   150→  }, [searchInput])
   151→
   152→  const load = useCallback(async () => {
   153→    setLoading(true)
   154→    setError(null)
   155→    try {
   156→      const params = new URLSearchParams({ year: String(year) })
   157→      if (quarter !== 'all') params.set('quarter', quarter)
   158→      if (search) params.set('q', search)
   159→      const res = await apiGet<{ invoices: PurchaseInvoice[] }>(`/api/purchase-invoices?${params.toString()}`)
   160→      setInvoices(res.invoices)
   161→    } catch (e) {
   162→      const message = e instanceof Error ? e.message : 'Không thể tải hóa đơn mua'
   163→      setError(message)
   164→    } finally {
   165→      setLoading(false)
   166→    }
   167→  }, [year, quarter, search])
   168→
   169→  useEffect(() => {
   170→    void load()
   171→  }, [load])
   172→
   173→  // Dữ liệu phụ cho dialog tạo: đối tác NCC + danh mục sản phẩm
   174→  const loadDialogData = useCallback(async () => {
   175→    try {
   176→      const [partnersRes, productsRes] = await Promise.all([
   177→        apiGet<{ partners: PartnerLite[] }>('/api/partners?type=SUPPLIER'),
   178→        apiGet<{ products: ProductLite[] }>('/api/products?status=ACTIVE'),
   179→      ])
   180→      setPartners(partnersRes.partners)
   181→      setProducts(productsRes.products)
   182→    } catch {
   183→      // bỏ qua — dialog vẫn mở, select sẽ rỗng
   184→    }
   185→  }, [])
   186→
   187→  const openCreate = () => {
   188→    setForm({ partnerId: 'none', invoiceDate: todayInput(), vatRate: '8', note: '' })
   189→    setRows([{ productId: '', quantity: '', unitPrice: '' }])
   190→    setDialogOpen(true)
   191→    void loadDialogData()
   192→  }
   193→
   194→  const updateRow = (index: number, patch: Partial<DraftRow>) =>
   195→    setRows((prev) => prev.map((row, i) => (i === index ? { ...row, ...patch } : row)))
   196→
   197→  const removeRow = (index: number) => setRows((prev) => prev.filter((_, i) => i !== index))
   198→
   199→  const addRow = () => setRows((prev) => [...prev, { productId: '', quantity: '', unitPrice: '' }])
   200→
   201→  // Quét mã vạch trong dialog tạo hóa đơn mua → thêm nhanh/tăng SL dòng sản phẩm
   202→  const handleScanToAdd = (code: string) => {
   203→    const product =
   204→      (products ?? []).find((p) => p.barcode === code) ?? (products ?? []).find((p) => p.sku === code)
   205→    if (!product) {
   206→      toast({ title: `Không tìm thấy sản phẩm với mã vạch ${code}`, variant: 'destructive' })
   207→      return
   208→    }
   209→    setRows((prev) => {
   210→      const existIdx = prev.findIndex((r) => r.productId === product.id)
   211→      if (existIdx >= 0) {
   212→        // Đã có dòng của SP này → tăng số lượng lên 1
   213→        return prev.map((row, i) =>
   214→          i === existIdx ? { ...row, quantity: String((Number(row.quantity) || 0) + 1) } : row
   215→        )
   216→      }
   217→      // Dòng đầu vẫn trống → điền vào thay vì thêm dòng mới
   218→      if (prev[0] && prev[0].productId === '') {
   219→        return prev.map((row, i) =>
   220→          i === 0 ? { productId: product.id, quantity: '1', unitPrice: String(product.costPrice) } : row
   221→        )
   222→      }
   223→      return [...prev, { productId: product.id, quantity: '1', unitPrice: String(product.costPrice) }]
   224→    })
   225→  }
   226→
   227→  const draftTotals = useMemo(() => {
   228→    const vatRate = Number(form.vatRate) || 0
   229→    let subtotal = 0
   230→    for (const row of rows) {
   231→      const qty = Number(row.quantity) || 0
   232→      const price = Number(row.unitPrice) || 0
   233→      subtotal += qty * price
   234→    }
   235→    const vatAmount = Math.round((subtotal * vatRate) / 100)
   236→    return { subtotal, vatAmount, total: subtotal + vatAmount }
   237→  }, [rows, form.vatRate])
   238→
   239→  const posted = (invoices ?? []).filter((inv) => inv.status === 'POSTED')
   240→  const totalExpense = posted.reduce((sum, inv) => sum + inv.total, 0)
   241→  const totalVatInput = posted.reduce((sum, inv) => sum + inv.vatAmount, 0)
   242→  const avgPerInvoice = posted.length > 0 ? Math.round(totalExpense / posted.length) : 0
   243→
   244→  const openDialogReady = dialogOpen && products !== null
   245→
   246→  const handleCreate = async () => {
   247→    const items: Array<{ productId: string; quantity: number; unitPrice?: number }> = []
   248→    for (const row of rows) {
   249→      const qty = Number(row.quantity)
   250→      const price = row.unitPrice === '' ? undefined : Number(row.unitPrice)
   251→      if (!row.productId) {
   252→        toast({ title: 'Vui lòng chọn sản phẩm cho mọi dòng', variant: 'destructive' })
   253→        return
   254→      }
   255→      if (!Number.isFinite(qty) || qty <= 0) {
   256→        toast({ title: 'Số lượng phải lớn hơn 0', variant: 'destructive' })
   257→        return
   258→      }
   259→      if (price !== undefined && (!Number.isFinite(price) || price < 0)) {
   260→        toast({ title: 'Đơn giá không được âm', variant: 'destructive' })
   261→        return
   262→      }
   263→      items.push({ productId: row.productId, quantity: qty, ...(price !== undefined ? { unitPrice: price } : {}) })
   264→    }
   265→    if (items.length === 0) {
   266→      toast({ title: 'Cần ít nhất 1 dòng sản phẩm', variant: 'destructive' })
   267→      return
   268→    }
   269→    setSubmitting(true)
   270→    try {
   271→      await apiPost<{ invoice: PurchaseInvoice }>('/api/purchase-invoices', {
   272→        partnerId: form.partnerId !== 'none' ? form.partnerId : undefined,
   273→        items,
   274→        vatRate: Number(form.vatRate) || 0,
   275→        invoiceDate: form.invoiceDate || undefined,
   276→        note: form.note.trim() || undefined,
   277→      })
   278→      toast({
   279→        title: 'Đã ghi sổ hóa đơn mua',
   280→        description: 'Chi phí & tồn kho đã được cập nhật theo hóa đơn.',
   281→      })
   282→      setDialogOpen(false)
   283→      await load()
   284→    } catch (e) {
   285→      toast({
   286→        title: 'Ghi sổ hóa đơn mua thất bại',
   287→        description: e instanceof Error ? e.message : 'Vui lòng thử lại',
   288→        variant: 'destructive',
   289→      })
   290→    } finally {
   291→      setSubmitting(false)
   292→    }
   293→  }
   294→
   295→  const handleCancel = async () => {
   296→    if (!cancelTarget) return
   297→    setCancelling(true)
   298→    try {
   299→      await apiPost<{ invoice: PurchaseInvoice }>(`/api/purchase-invoices/${cancelTarget.id}`, { action: 'CANCEL' })
   300→      toast({
   301→        title: 'Đã hủy hóa đơn mua',
   302→        description: `${cancelTarget.code} — tồn kho & chi phí đã hoàn ngược.`,
   303→      })
   304→      setCancelTarget(null)
   305→      setDetail(null)
   306→      await load()
   307→    } catch (e) {
   308→      toast({
   309→        title: 'Hủy hóa đơn thất bại',
   310→        description: e instanceof Error ? e.message : 'Vui lòng thử lại',
   311→        variant: 'destructive',
   312→      })
   313→    } finally {
   314→      setCancelling(false)
   315→    }
   316→  }
   317→
   318→  const selectProduct = (index: number, productId: string) => {
   319→    const product = (products ?? []).find((p) => p.id === productId)
   320→    updateRow(index, {
   321→      productId,
   322→      unitPrice:
   323→        product && (rows[index].unitPrice === '' || Number(rows[index].unitPrice) === 0)
   324→          ? String(product.costPrice)
   325→          : rows[index].unitPrice,
   326→    })
   327→  }
   328→
   329→  return (
   330→    <div>
   331→      <PageHeader
   332→        title="Nhập hàng"
   333→        description="Hóa đơn mua nguyên vật liệu — ghi sổ chi phí, thuế VAT đầu vào và nhập kho"
   334→      >
   335→        <Button onClick={openCreate}>
   336→          <PackagePlus className="h-4 w-4" />
   337→          Tạo hóa đơn mua
   338→        </Button>
   339→      </PageHeader>
   340→
   341→      <div className="mb-6 grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
   342→        <StatCard
   343→          title="Tổng chi"
   344→          value={loading ? '' : formatCompactVND(totalExpense)}
   345→          icon={Wallet}
   346→          loading={loading}
   347→          hint="Đã loại hóa đơn hủy"
   348→          iconClassName="bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-400"
   349→        />
   350→        <StatCard
   351→          title="Số hóa đơn"
   352→          value={loading ? '' : formatNumber(invoices?.length ?? 0)}
   353→          icon={Receipt}
   354→          loading={loading}
   355→          hint={`${posted.length} đã ghi sổ`}
   356→        />
   357→        <StatCard
   358→          title="Thuế VAT đầu vào"
   359→          value={loading ? '' : formatCompactVND(totalVatInput)}
   360→          icon={FileSpreadsheet}
   361→          loading={loading}
   362→          hint="Khấu trừ từ hóa đơn mua"
   363→          iconClassName="bg-sky-100 text-sky-700 dark:bg-sky-950 dark:text-sky-400"
   364→        />
   365→        <StatCard
   366→          title="Giá trị TB / hóa đơn"
   367→          value={loading ? '' : formatCompactVND(avgPerInvoice)}
   368→          icon={PackagePlus}
   369→          loading={loading}
   370→          hint="Trên hóa đơn đã ghi sổ"
   371→        />
   372→      </div>
   373→
   374→      {/* Bộ lọc */}
   375→      <div className="mb-4 flex flex-wrap items-center gap-3">
   376→        <Select value={quarter} onValueChange={setQuarter}>
   377→          <SelectTrigger className="w-[140px]">
   378→            <SelectValue />
   379→          </SelectTrigger>
   380→          <SelectContent>
   381→            {QUARTER_OPTIONS.map((q) => (
   382→              <SelectItem key={q.value} value={q.value}>
   383→                {q.label}
   384→              </SelectItem>
   385→            ))}
   386→          </SelectContent>
   387→        </Select>
   388→        <Select value={String(year)} onValueChange={(value) => setYear(Number(value))}>
   389→          <SelectTrigger className="w-[120px]">
   390→            <SelectValue />
   391→          </SelectTrigger>
   392→          <SelectContent>
   393→            {YEAR_OPTIONS.map((y) => (
   394→              <SelectItem key={y} value={String(y)}>
   395→                Năm {y}
   396→              </SelectItem>
   397→            ))}
   398→          </SelectContent>
   399→        </Select>
   400→        <div className="relative w-full max-w-xs">
   401→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   402→          <Input
   403→            value={searchInput}
   404→            onChange={(e) => setSearchInput(e.target.value)}
   405→            placeholder="Tìm mã hóa đơn, đối tác..."
   406→            className="pl-8"
   407→          />
   408→        </div>
   409→      </div>
   410→
   411→      <Card className="border-border/70 shadow-sm">
   412→        <CardContent className="p-0">
   413→          {loading ? (
   414→            <div className="p-4">
   415→              <TableSkeleton rows={6} />
   416→            </div>
   417→          ) : error ? (
   418→            <div className="p-4">
   419→              <ErrorState message={error} onRetry={() => void load()} />
   420→            </div>
   421→          ) : !invoices || invoices.length === 0 ? (
   422→            <div className="p-4">
   423→              <EmptyState
   424→                icon={PackagePlus}
   425→                title="Chưa có hóa đơn mua nào"
   426→                description="Tạo hóa đơn mua để ghi sổ chi phí và nhập kho nguyên vật liệu."
   427→                action={
   428→                  <Button onClick={openCreate}>
   429→                    <PackagePlus className="h-4 w-4" />
   430→                    Tạo hóa đơn mua
   431→                  </Button>
   432→                }
   433→              />
   434→            </div>
   435→          ) : (
   436→            <div className="max-h-96 overflow-auto">
   437→              <Table>
   438→                <TableHeader className="sticky top-0 bg-background">
   439→                  <TableRow>
   440→                    <TableHead>Mã HĐ</TableHead>
   441→                    <TableHead>Ngày</TableHead>
   442→                    <TableHead>Đối tác</TableHead>
   443→                    <TableHead className="text-right">SL mặt hàng</TableHead>
   444→                    <TableHead className="text-right">Tiền hàng</TableHead>
   445→                    <TableHead className="text-right">VAT</TableHead>
   446→                    <TableHead className="text-right">Tổng cộng</TableHead>
   447→                    <TableHead>Trạng thái</TableHead>
   448→                    <TableHead className="w-12" />
   449→                  </TableRow>
   450→                </TableHeader>
   451→                <TableBody>
   452→                  {invoices.map((inv) => (
   453→                    <TableRow key={inv.id} className={inv.status === 'CANCELLED' ? 'opacity-60' : undefined}>
   454→                      <TableCell className="whitespace-nowrap font-mono text-xs font-medium">{inv.code}</TableCell>
   455→                      <TableCell className="whitespace-nowrap text-sm">{formatDate(inv.invoiceDate)}</TableCell>
   456→                      <TableCell>
   457→                        <p className="max-w-[180px] truncate font-medium">{inv.partner?.name ?? '—'}</p>
   458→                        {inv.partner?.taxCode && (
   459→                          <p className="font-mono text-xs text-muted-foreground">{inv.partner.taxCode}</p>
   460→                        )}
   461→                      </TableCell>
   462→                      <TableCell className="text-right tabular-nums">{inv.items.length}</TableCell>
   463→                      <TableCell className="whitespace-nowrap text-right tabular-nums">{formatVND(inv.subtotal)}</TableCell>
   464→                      <TableCell className="whitespace-nowrap text-right tabular-nums text-sky-700 dark:text-sky-400">
   465→                        {inv.vatRate}% · {formatVND(inv.vatAmount)}
   466→                      </TableCell>
   467→                      <TableCell className="whitespace-nowrap text-right font-semibold tabular-nums">
   468→                        {formatVND(inv.total)}
   469→                      </TableCell>
   470→                      <TableCell>
   471→                        {inv.status === 'POSTED' ? (
   472→                          <Badge className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400">
   473→                            Đã ghi sổ
   474→                          </Badge>
   475→                        ) : (
   476→                          <Badge variant="secondary">Đã hủy</Badge>
   477→                        )}
   478→                      </TableCell>
   479→                      <TableCell>
   480→                        <Button
   481→                          variant="ghost"
   482→                          size="icon"
   483→                          className="h-8 w-8"
   484→                          onClick={() => setDetail(inv)}
   485→                          title="Xem chi tiết"
   486→                        >
   487→                          <Eye className="h-4 w-4" />
   488→                        </Button>
   489→                      </TableCell>
   490→                    </TableRow>
   491→                  ))}
   492→                </TableBody>
   493→              </Table>
   494→            </div>
   495→          )}
   496→        </CardContent>
   497→      </Card>
   498→
   499→      {/* Dialog chi tiết hóa đơn */}
   500→      <Dialog open={detail !== null} onOpenChange={(open) => !open && setDetail(null)}>
   501→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-2xl">
   502→          {detail && (
   503→            <>
   504→              <DialogHeader>
   505→                <DialogTitle className="flex flex-wrap items-center gap-2">
   506→                  <span className="font-mono">{detail.code}</span>
   507→                  {detail.status === 'POSTED' ? (
   508→                    <Badge className="border-transparent bg-emerald-100 text-emerald-700 dark:bg-emerald-950 dark:text-emerald-400">
   509→                      Đã ghi sổ
   510→                    </Badge>
   511→                  ) : (
   512→                    <Badge variant="secondary">Đã hủy</Badge>
   513→                  )}
   514→                </DialogTitle>
   515→                <DialogDescription>
   516→                  {formatDate(detail.invoiceDate)} — {detail.partner?.name ?? 'Không có đối tác'}
   517→                  {detail.partner?.taxCode ? ` (MST ${detail.partner.taxCode})` : ''}
   518→                </DialogDescription>
   519→              </DialogHeader>
   520→              <div className="overflow-x-auto">
   521→                <Table>
   522→                  <TableHeader>
   523→                    <TableRow>
   524→                      <TableHead className="font-mono">Mã SP</TableHead>
