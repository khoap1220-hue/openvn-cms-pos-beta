    88→function errorMessage(err: unknown): string {
    89→  return err instanceof Error ? err.message : 'Đã xảy ra lỗi'
    90→}
    91→
    92→// Các trạng thái đích hợp lệ từ trạng thái hiện tại (chỉ tiến theo luồng + có thể hủy)
    93→function allowedNextStatuses(current: string): string[] {
    94→  if (current === 'CANCELLED') return []
    95→  const idx = STATUS_FLOW.indexOf(current as (typeof STATUS_FLOW)[number])
    96→  const forward = idx >= 0 ? STATUS_FLOW.slice(idx + 1) : STATUS_FLOW
    97→  return [...forward, 'CANCELLED']
    98→}
    99→
   100→function itemsSummary(order: Order): string {
   101→  if (order.items.length === 0) return '—'
   102→  if (order.items.length === 1) {
   103→    const item = order.items[0]
   104→    return `${item.quantity} × ${item.product?.name ?? 'Sản phẩm'}`
   105→  }
   106→  return `${order.items.length} sản phẩm`
   107→}

   114→    </div>
   115→  )
   116→}

   295→        title: 'Không thể cập nhật thanh toán',
   296→        description: errorMessage(err),
   297→        variant: 'destructive',

   301→    }
   302→  }
   303→

   313→

   332→        if (!cancelled) setCreateLoadingProducts(false)

   335→      cancelled = true
   336→    }
   337→  }, [createOpen, toast])

   343→      (p) => p.name.toLowerCase().includes(keyword) || p.sku.toLowerCase().includes(keyword)

   346→
   347→  const createTotal = useMemo(
   348→    () => items.reduce((sum, item) => sum + item.price * item.quantity, 0),
   349→    [items]

   351→
   352→  const handleAddItem = () => {

   354→    if (!product) {
   355→      toast({ title: 'Vui lòng chọn một sản phẩm', variant: 'destructive' })

   357→    }

   359→    setItems((prev) => {
   360→      const existing = prev.find((i) => i.productId === product.id)
   361→      if (existing) {
   362→        return prev.map((i) => (i.productId === product.id ? { ...i, quantity: i.quantity + qty } : i))
   363→      }

   365→        ...prev,
   366→        { productId: product.id, name: product.name, sku: product.sku, price: product.sellPrice, quantity: qty },

   375→      prev.map((i) =>
   376→        i.productId === productId ? { ...i, quantity: Math.max(1, i.quantity + delta) } : i
   377→      )
   378→    )
   379→  }
   380→

   382→    setItems((prev) => prev.filter((i) => i.productId !== productId))
   383→  }
   384→
   385→  const submitCreate = async () => {

   388→      return
   389→    }
   390→    if (!createForm.customerName.trim()) {
   391→      toast({ title: 'Vui lòng nhập tên khách hàng', variant: 'destructive' })
   392→      return
   393→    }

   395→      toast({ title: 'Vui lòng thêm ít nhất một sản phẩm vào đơn', variant: 'destructive' })

   412→      setRefreshKey((k) => k + 1)

   414→      // Lỗi 400: hết hàng hoặc dữ liệu không hợp lệ — hiển thị message từ server
   415→      toast({
   416→        title: 'Không thể tạo đơn hàng',
   417→        description: errorMessage(err),
   418→        variant: 'destructive',
   419→      })
   420→    } finally {
   421→      setSubmitting(false)
   422→    }
   423→  }
   424→
   425→  // ===== Phân trang hiển thị =====
   426→  const from = total === 0 ? 0 : (page - 1) * PAGE_SIZE + 1
   427→  const to = Math.min(page * PAGE_SIZE, total)
   428→  const hasFilter = search !== '' || statusFilter !== 'ALL' || channelFilter !== 'ALL'
   429→  const nextStatuses = detail ? allowedNextStatuses(detail.status) : []
   430→
   431→  return (

   439→      {/* Toolbar */}

   441→        <div className="relative w-full sm:w-64">

   444→            value={searchInput}
   445→            onChange={(e) => setSearchInput(e.target.value)}
   446→            placeholder="Tìm mã đơn, tên hoặc SĐT khách..."
   447→            className="pl-8"
   448→            aria-label="Tìm kiếm đơn hàng"
   449→          />
   450→        </div>
   451→        <Select
   452→          value={statusFilter}
   453→          onValueChange={(v) => {
   454→            setStatusFilter(v)
   455→            setPage(1)
   456→          }}
   457→        >
   458→          <SelectTrigger className="w-full sm:w-[170px]" aria-label="Lọc theo trạng thái">
   459→            <SelectValue placeholder="Tất cả trạng thái" />
   460→          </SelectTrigger>
   461→          <SelectContent>
   462→            <SelectItem value="ALL">Tất cả trạng thái</SelectItem>
   463→            {ORDER_STATUSES.map((s) => (
   464→              <SelectItem key={s} value={s}>
   465→                {ORDER_STATUS_LABELS[s] ?? s}
   466→              </SelectItem>
   467→            ))}
   468→          </SelectContent>
   469→        </Select>
   470→        <Select
   471→          value={channelFilter}
   472→          onValueChange={(v) => {
   473→            setChannelFilter(v)
   474→            setPage(1)
   475→          }}
   476→        >
   477→          <SelectTrigger className="w-full sm:w-[170px]" aria-label="Lọc theo kênh bán">
   478→            <SelectValue placeholder="Tất cả kênh" />
   479→          </SelectTrigger>
   480→          <SelectContent>
   481→            <SelectItem value="ALL">Tất cả kênh</SelectItem>
   482→            {channels.map((c) => (
   483→              <SelectItem key={c.id} value={c.id}>
   484→                {c.name}
   485→              </SelectItem>
   486→            ))}
   487→          </SelectContent>
   488→        </Select>
   489→      </div>
   490→
   491→      {/* Nội dung chính */}
   492→      {loading ? (
   493→        <TableSkeleton rows={7} />
   494→      ) : error ? (
   495→        <ErrorState message={error} onRetry={() => void fetchOrders()} />
   496→      ) : orders.length === 0 ? (
   497→        <EmptyState
   498→          icon={ShoppingCart}
   499→          title={hasFilter ? 'Không tìm thấy đơn hàng nào' : 'Chưa có đơn hàng nào'}
   500→          description={
   501→            hasFilter
   502→              ? 'Thử thay đổi từ khóa tìm kiếm hoặc bộ lọc.'
   503→              : 'Các đơn hàng từ mọi kênh bán sẽ hiển thị tại đây.'
   504→          }
   505→          action={
   506→            <Button onClick={openCreate}>
   507→              <Plus className="mr-1.5 h-4 w-4" /> Tạo đơn
   508→            </Button>
   509→          }
   510→        />
   511→      ) : (
   512→        <>
   513→          <div className="overflow-x-auto rounded-lg border">
   514→            <Table className="min-w-[960px]">
   515→              <TableHeader>
   516→                <TableRow>
   517→                  <TableHead>Mã đơn</TableHead>
   518→                  <TableHead>Kênh</TableHead>
   519→                  <TableHead>Khách hàng</TableHead>
   520→                  <TableHead>Sản phẩm</TableHead>
   521→                  <TableHead className="text-right">Tổng tiền</TableHead>
   522→                  <TableHead>Thanh toán</TableHead>
   523→                  <TableHead>Trạng thái</TableHead>
   524→                  <TableHead>Thời gian</TableHead>
   525→                  <TableHead className="text-right">Hành động</TableHead>
   526→                </TableRow>
   527→              </TableHeader>
   528→              <TableBody>
   529→                {orders.map((o) => (

   532→                    <TableCell>
   533→                      <ChannelBadge channel={o.channel} className="text-xs" />
   534→                    </TableCell>
   535→                    <TableCell>
   536→                      <p className="text-sm font-medium">{o.customer?.name ?? 'Khách lẻ'}</p>

   538→                        <p className="text-xs text-muted-foreground">{o.customer.phone}</p>

   540→                    </TableCell>
   541→                    <TableCell className="max-w-[180px] truncate text-sm">{itemsSummary(o)}</TableCell>
   542→                    <TableCell className="whitespace-nowrap text-right font-semibold">
   543→                      {formatVND(o.total)}
   544→                    </TableCell>
   545→                    <TableCell>
   546→                      <PaymentStatusBadge status={o.paymentStatus} />
   547→                    </TableCell>
   548→                    <TableCell>
   549→                      <OrderStatusBadge status={o.status} />
   550→                    </TableCell>
   551→                    <TableCell className="whitespace-nowrap text-sm text-muted-foreground">
