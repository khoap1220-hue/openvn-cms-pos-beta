    73→    73→  categoryId: string

    75→    75→  sellPrice: string

    98→    98→
    99→    99→export function ProductsView() {
   100→   100→  const { toast } = useToast()
   101→   101→
   102→   102→  // ===== Danh sách sản phẩm + bộ lọc =====
   103→   103→  const [products, setProducts] = useState<Product[]>([])
   104→   104→  const [loading, setLoading] = useState(true)
   105→   105→  const [error, setError] = useState<string | null>(null)
   106→   106→
   107→   107→  const [searchInput, setSearchInput] = useState('')
   108→   108→  const [search, setSearch] = useState('')
   109→   109→  const [categoryFilter, setCategoryFilter] = useState('ALL')
   110→   110→  const [statusFilter, setStatusFilter] = useState('ALL')
   111→   111→  const [page, setPage] = useState(1)
   112→   112→
   113→   113→  // ===== Meta data =====
   114→   114→  const [categories, setCategories] = useState<Category[]>([])
   115→   115→  const [channels, setChannels] = useState<Channel[]>([])
   116→   116→
   117→   117→  // ===== Dialog thêm / sửa =====
   118→   118→  const [dialogOpen, setDialogOpen] = useState(false)
   119→   119→  const [editing, setEditing] = useState<Product | null>(null)
   120→   120→  const [form, setForm] = useState<ProductFormState>(EMPTY_FORM)
   121→   121→  const [saving, setSaving] = useState(false)
   122→   122→
   123→   123→  // ===== Xóa sản phẩm =====
   124→   124→  const [deleting, setDeleting] = useState<Product | null>(null)
   125→   125→  const [deleteLoading, setDeleteLoading] = useState(false)
   126→   126→
   127→   127→  // ===== Tìm kiếm debounce (client-side) =====
   128→   128→  useEffect(() => {
   129→   129→    const timer = setTimeout(() => {
   130→   130→      setSearch(searchInput.trim())
   131→   131→      setPage(1)
   132→   132→    }, 400)
   133→   133→    return () => clearTimeout(timer)
   134→   134→  }, [searchInput])
   135→   135→
   136→   136→  const fetchProducts = useCallback(async () => {
   137→   137→    setLoading(true)
   138→   138→    setError(null)
   139→   139→    try {
   140→   140→      const params = new URLSearchParams()
   141→   141→      if (search) params.set('search', search)
   142→   142→      if (categoryFilter !== 'ALL') params.set('categoryId', categoryFilter)
   143→   143→      if (statusFilter !== 'ALL') params.set('status', statusFilter)
   144→   144→      const qs = params.toString()
   145→   145→      const data = await apiGet<{ products: Product[] }>(`/api/products${qs ? `?${qs}` : ''}`)
   146→   146→      setProducts(data.products)
   147→   147→    } catch (err) {
   148→   148→      setError(errorMessage(err))
   149→   149→    } finally {
   150→   150→      setLoading(false)
   151→   151→    }
   152→   152→  }, [search, categoryFilter, statusFilter])
   153→   153→
   154→   154→  useEffect(() => {
   155→   155→    void fetchProducts()
   156→   156→  }, [fetchProducts])
   157→   157→
   158→   158→  const fetchMeta = useCallback(async () => {
   159→   159→    try {
   160→   160→      const [catRes, chRes] = await Promise.all([
   161→   161→        apiGet<{ categories: Category[] }>('/api/categories'),
   162→   162→        apiGet<{ channels: Channel[] }>('/api/channels'),
   163→   163→      ])
   164→   164→      setCategories(catRes.categories)
   165→   165→      setChannels(chRes.channels)
   166→   166→    } catch (err) {
   167→   167→      toast({
   168→   168→        title: 'Không thể tải danh mục / kênh bán',
   169→   169→        description: errorMessage(err),
   170→   170→        variant: 'destructive',
   171→   171→      })
   172→   172→    }
   173→   173→  }, [toast])
   174→   174→
   175→   175→  useEffect(() => {
   176→   176→    void fetchMeta()
   177→   177→  }, [fetchMeta])
   178→   178→
   179→   179→  // ===== Phân trang client-side =====
   180→   180→  const totalPages = Math.max(1, Math.ceil(products.length / PAGE_SIZE))
   181→   181→  const currentPage = Math.min(page, totalPages)
   182→   182→  const pagedProducts = useMemo(
   183→   183→    () => products.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE),
   184→   184→    [products, currentPage]
   185→   185→  )
   186→   186→  const hasFilter = search !== '' || categoryFilter !== 'ALL' || statusFilter !== 'ALL'
   187→   187→
   188→   188→  // ===== Dialog handlers =====
   189→   189→  const openCreate = () => {
   190→   190→    setEditing(null)
   191→   191→    setForm(EMPTY_FORM)
   192→   192→    setDialogOpen(true)
   193→   193→  }
   194→   194→
   195→   195→  const openEdit = (p: Product) => {
   196→   196→    setEditing(p)
   197→   197→    setForm({
   198→   198→      sku: p.sku,
   199→   199→      name: p.name,
   200→   200→      description: p.description ?? '',
   201→   201→      categoryId: p.categoryId ?? 'NONE',
   202→   202→      costPrice: String(p.costPrice),
   203→   203→      sellPrice: String(p.sellPrice),
   204→   204→      stock: String(p.stock),
   205→   205→      lowStockThreshold: String(p.lowStockThreshold),
   206→   206→      status: p.status,
   207→   207→      listingChannelIds: (p.listings ?? []).map((l) => l.channelId),
   208→   208→    })
   209→   209→    setDialogOpen(true)
   210→   210→  }
   211→   211→
   212→   212→  const toggleChannel = (channelId: string, checked: boolean) => {
   213→   213→    setForm((f) => ({
   214→   214→      ...f,
   215→   215→      listingChannelIds: checked
   216→   216→        ? [...f.listingChannelIds, channelId]
   217→   217→        : f.listingChannelIds.filter((id) => id !== channelId),
   218→   218→    }))
   219→   219→  }
   220→   220→
   221→   221→  const handleSave = async () => {
   222→   222→    if (!form.sku.trim()) {
   223→   223→      toast({ title: 'Vui lòng nhập SKU sản phẩm', variant: 'destructive' })
   224→   224→      return
   225→   225→    }
   226→   226→    if (!form.name.trim()) {
   227→   227→      toast({ title: 'Vui lòng nhập tên sản phẩm', variant: 'destructive' })
   228→   228→      return
   229→   229→    }
   230→   230→    if (!Number(form.sellPrice)) {
   231→   231→      toast({ title: 'Vui lòng nhập giá bán hợp lệ', variant: 'destructive' })
   232→   232→      return
   233→   233→    }
   234→   234→    setSaving(true)
   235→   235→    try {
   236→   236→      const base = {
   237→   237→        sku: form.sku.trim(),
   238→   238→        name: form.name.trim(),
   239→   239→        description: form.description.trim(),
   240→   240→        categoryId: form.categoryId === 'NONE' ? null : form.categoryId,
   241→   241→        costPrice: Number(form.costPrice) || 0,
   242→   242→        sellPrice: Number(form.sellPrice) || 0,
   243→   243→        lowStockThreshold: Number(form.lowStockThreshold) || 0,
   244→   244→        status: form.status,
   245→   245→        listingChannelIds: form.listingChannelIds,
   246→   246→      }
   247→   247→      if (editing) {
   248→   248→        await apiPut<{ product: Product }>(`/api/products/${editing.id}`, base)
   249→   249→        toast({ title: 'Đã cập nhật sản phẩm', description: base.name })
   250→   250→      } else {
   251→   251→        await apiPost<{ product: Product }>('/api/products', { ...base, stock: Number(form.stock) || 0 })
   252→   252→        toast({ title: 'Đã thêm sản phẩm mới', description: base.name })
   253→   253→      }
   254→   254→      setDialogOpen(false)
   255→   255→      void fetchProducts()
   256→   256→    } catch (err) {
   257→   257→      toast({
   258→   258→        title: editing ? 'Không thể cập nhật sản phẩm' : 'Không thể thêm sản phẩm',
   259→   259→        description: errorMessage(err),
   260→   260→        variant: 'destructive',
   261→   261→      })
   262→   262→    } finally {
   263→   263→      setSaving(false)
   264→   264→    }
   265→   265→  }
   266→   266→
   267→   267→  const handleToggleStatus = async (p: Product) => {
   268→   268→    const nextStatus = p.status === 'ACTIVE' ? 'INACTIVE' : 'ACTIVE'
   269→   269→    try {
   270→   270→      await apiPut<{ product: Product }>(`/api/products/${p.id}`, { status: nextStatus })
   271→   271→      toast({
   272→   272→        title: nextStatus === 'ACTIVE' ? 'Đã kích hoạt lại sản phẩm' : 'Đã ngừng bán sản phẩm',
   273→   273→        description: p.name,
   274→   274→      })
   275→   275→      void fetchProducts()
   276→   276→    } catch (err) {
   277→   277→      toast({
   278→   278→        title: 'Cập nhật trạng thái thất bại',
   279→   279→        description: errorMessage(err),
   280→   280→        variant: 'destructive',
   281→   281→      })
   282→   282→    }
   283→   283→  }
   284→   284→
   285→   285→  const handleDelete = async () => {
   286→   286→    if (!deleting) return
   287→   287→    setDeleteLoading(true)
   288→   288→    try {
   289→   289→      await apiDelete<{ success: boolean }>(`/api/products/${deleting.id}`)
   290→   290→      toast({ title: 'Đã xóa sản phẩm', description: deleting.name })
   291→   291→      setDeleting(null)
   292→   292→      void fetchProducts()
   293→   293→    } catch (err) {
   294→   294→      // 409: sản phẩm đã có đơn hàng → gợi ý chuyển sang "Ngừng bán"
   295→   295→      toast({
   296→   296→        title: 'Không thể xóa sản phẩm',
   297→   297→        description: `${errorMessage(err)}. Nếu sản phẩm đã có đơn hàng, hãy chuyển sang "Ngừng bán" thay vì xóa.`,
   298→   298→        variant: 'destructive',
   299→   299→      })
   300→   300→      setDeleting(null)
   301→   301→    } finally {
   302→   302→      setDeleteLoading(false)
   303→   303→    }
   304→   304→  }
   305→   305→
   306→   306→  return (
   307→   307→    <div>
   308→   308→      <PageHeader title="Sản phẩm" description="Quản lý sản phẩm và kênh phân phối đa kênh">
   309→   309→        <Button onClick={openCreate}>
   310→   310→          <Plus className="mr-1.5 h-4 w-4" /> Thêm sản phẩm
   311→   311→        </Button>
   312→   312→      </PageHeader>
   313→   313→
   314→   314→      {/* Toolbar */}
   315→   315→      <div className="mb-4 flex flex-col gap-2 sm:flex-row sm:items-center">
   316→   316→        <div className="relative w-full sm:w-72">
   317→   317→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   318→   318→          <Input
   319→   319→            value={searchInput}
   320→   320→            onChange={(e) => setSearchInput(e.target.value)}
   321→   321→            placeholder="Tìm theo tên hoặc SKU..."
   322→   322→            className="pl-8"
   323→   323→            aria-label="Tìm kiếm sản phẩm"
   324→   324→          />
   325→   325→        </div>
   326→   326→        <Select
   327→   327→          value={categoryFilter}
   328→   328→          onValueChange={(v) => {
   329→   329→            setCategoryFilter(v)
   330→   330→            setPage(1)
   331→   331→          }}
   332→   332→        >
   333→   333→          <SelectTrigger className="w-full sm:w-[180px]" aria-label="Lọc theo danh mục">
   334→   334→            <SelectValue placeholder="Tất cả danh mục" />
   335→   335→          </SelectTrigger>
   336→   336→          <SelectContent>
   337→   337→            <SelectItem value="ALL">Tất cả danh mục</SelectItem>
   338→   338→            {categories.map((c) => (
   339→   339→              <SelectItem key={c.id} value={c.id}>
   340→   340→                {c.name}
   341→   341→              </SelectItem>
   342→   342→            ))}
   343→   343→          </SelectContent>
   344→   344→        </Select>
   345→   345→        <Select
   346→   346→          value={statusFilter}
   347→   347→          onValueChange={(v) => {
   348→   348→            setStatusFilter(v)
   349→   349→            setPage(1)
   350→   350→          }}
   351→   351→        >
   352→   352→          <SelectTrigger className="w-full sm:w-[150px]" aria-label="Lọc theo trạng thái">
   353→   353→            <SelectValue placeholder="Tất cả trạng thái" />
   354→   354→          </SelectTrigger>
   355→   355→          <SelectContent>
   356→   356→            <SelectItem value="ALL">Tất cả trạng thái</SelectItem>
   357→   357→            <SelectItem value="ACTIVE">Đang bán</SelectItem>
   358→   358→            <SelectItem value="INACTIVE">Ngừng bán</SelectItem>
   359→   359→          </SelectContent>
   360→   360→        </Select>
   361→   361→      </div>
   362→   362→
   363→   363→      {/* Nội dung chính */}
   364→   364→      {loading ? (
   365→   365→        <TableSkeleton rows={6} />
   366→   366→      ) : error ? (
   367→   367→        <ErrorState message={error} onRetry={() => void fetchProducts()} />
   368→   368→      ) : products.length === 0 ? (
   369→   369→        <EmptyState
   370→   370→          icon={Package}
   371→   371→          title={hasFilter ? 'Không tìm thấy sản phẩm nào' : 'Chưa có sản phẩm nào'}
   372→   372→          description={
   373→   373→            hasFilter
   374→   374→              ? 'Thử thay đổi từ khóa tìm kiếm hoặc bộ lọc.'
   375→   375→              : 'Bắt đầu bằng cách thêm sản phẩm đầu tiên cho cửa hàng.'
   376→   376→          }
   377→   377→          action={
   378→   378→            <Button onClick={openCreate}>
   379→   379→              <Plus className="mr-1.5 h-4 w-4" /> Thêm sản phẩm
   380→   380→            </Button>
   381→   381→          }
   382→   382→        />
   383→   383→      ) : (
   384→   384→        <>
   385→   385→          <div className="overflow-x-auto rounded-lg border">
   386→   386→            <Table className="min-w-[900px]">
   387→   387→              <TableHeader>
   388→   388→                <TableRow>
   389→   389→                  <TableHead>Sản phẩm</TableHead>
   390→   390→                  <TableHead>Danh mục</TableHead>
   391→   391→                  <TableHead>Giá bán</TableHead>
   392→   392→                  <TableHead>Tồn kho</TableHead>
   393→   393→                  <TableHead>Kênh bán</TableHead>
   394→   394→                  <TableHead>Trạng thái</TableHead>
   395→   395→                  <TableHead className="w-12 text-right">
   396→   396→                    <span className="sr-only">Hành động</span>
   397→   397→                  </TableHead>
   398→   398→                </TableRow>
   399→   399→              </TableHeader>
   400→   400→              <TableBody>
   401→   401→                {pagedProducts.map((p) => (
   402→   402→                  <TableRow key={p.id}>
   403→   403→                    <TableCell>
   404→   404→                      <div className="flex items-center gap-3">
   405→   405→                        <ProductAvatar name={p.name} />
   406→   406→                        <div className="min-w-0">
   407→   407→                          <p className="max-w-[220px] truncate font-medium">{p.name}</p>
   408→   408→                          <p className="text-xs text-muted-foreground">{p.sku}</p>
   409→   409→                        </div>
   410→   410→                      </div>
   411→   411→                    </TableCell>
   412→   412→                    <TableCell className="text-sm">{p.category?.name ?? '—'}</TableCell>
   413→   413→                    <TableCell className="whitespace-nowrap text-sm">{formatVND(p.sellPrice)}</TableCell>
   414→   414→                    <TableCell>
   415→   415→                      {p.stock <= p.lowStockThreshold ? (
   416→   416→                        <div className="space-y-0.5">
   417→   417→                          <Badge
   418→   418→                            variant="outline"
   419→   419→                            className="border-red-200 bg-red-100 text-red-800 dark:border-red-900 dark:bg-red-950 dark:text-red-400"
   420→   420→                          >
   421→   421→                            {p.stock === 0 ? 'Hết hàng' : `Còn ${p.stock}`}
   422→   422→                          </Badge>
   423→   423→                          <p className="text-xs text-muted-foreground">Ngưỡng {p.lowStockThreshold}</p>
   424→   424→                        </div>
   425→   425→                      ) : (
   426→   426→                        <span className="text-sm">{formatNumber(p.stock)}</span>
   427→   427→                      )}
   428→   428→                    </TableCell>
   429→   429→                    <TableCell>
   430→   430→                      <div className="flex max-w-[160px] flex-wrap gap-x-3 gap-y-1">
   431→   431→                        {(p.listings ?? []).length > 0 ? (
   432→   432→                          (p.listings ?? []).map((l) => (
   433→   433→                            <ChannelBadge key={l.id} channel={l.channel} className="text-xs" />
   434→   434→                          ))
   435→   435→                        ) : (
   436→   436→                          <span className="text-xs text-muted-foreground">—</span>
   437→   437→                        )}
   438→   438→                      </div>
   439→   439→                    </TableCell>
   440→   440→                    <TableCell>
   441→   441→                      <ProductStatusBadge status={p.status} />
   442→   442→                    </TableCell>
   443→   443→                    <TableCell className="text-right">
   444→   444→                      <DropdownMenu>
   445→   445→                        <DropdownMenuTrigger asChild>
   446→   446→                          <Button variant="ghost" size="icon" className="h-8 w-8">
   447→   447→                            <MoreHorizontal className="h-4 w-4" />
   448→   448→                            <span className="sr-only">Hành động cho {p.name}</span>
   449→   449→                          </Button>
   450→   450→                        </DropdownMenuTrigger>
   451→   451→                        <DropdownMenuContent align="end">
   452→   452→                          <DropdownMenuItem onSelect={() => openEdit(p)}>
   453→   453→                            <Pencil className="mr-2 h-4 w-4" /> Sửa
   454→   454→                          </DropdownMenuItem>
   455→   455→                          <DropdownMenuItem onSelect={() => void handleToggleStatus(p)}>
   456→   456→                            <Power className="mr-2 h-4 w-4" />
   457→   457→                            {p.status === 'ACTIVE' ? 'Ngừng bán' : 'Kích hoạt lại'}
