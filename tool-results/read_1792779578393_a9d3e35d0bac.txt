   105→function errorMessage(err: unknown): string {

   109→// Các trạng thái đích hợp lệ từ trạng thái hiện tại (chỉ tiến theo luồng + có thể hủy)
   110→function allowedNextStatuses(current: string): string[] {
   111→  if (current === 'CANCELLED') return []
   112→  const idx = STATUS_FLOW.indexOf(current as (typeof STATUS_FLOW)[number])
   113→  const forward = idx >= 0 ? STATUS_FLOW.slice(idx + 1) : STATUS_FLOW
   114→  return [...forward, 'CANCELLED']
   115→}

   117→function itemsSummary(order: Order): string {
   118→  if (order.items.length === 0) return '—'
   119→  if (order.items.length === 1) {
   120→    const item = order.items[0]
   121→    return `${item.quantity} × ${item.product?.name ?? 'Sản phẩm'}`
   122→  }
   123→  return `${order.items.length} sản phẩm`
   124→}
