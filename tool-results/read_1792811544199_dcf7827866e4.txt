   109→   109→export function ProductsView() {
   110→   110→  const { toast } = useToast()
   111→   111→
   112→   112→  // ===== Danh sách sản phẩm + bộ lọc =====
   113→   113→  const [products, setProducts] = useState<ProductRow[]>([])
   114→   114→  const [loading, setLoading] = useState(true)
   115→   115→  const [error, setError] = useState<string | null>(null)
   116→   116→
   117→   117→  const [searchInput, setSearchInput] = useState('')
   118→   118→  const [search, setSearch] = useState('')
   119→   119→  const [categoryFilter, setCategoryFilter] = useState('ALL')
   120→   120→  const [statusFilter, setStatusFilter] = useState('ALL')
   121→   121→  const [page, setPage] = useState(1)
   122→   122→
   123→   123→  // ===== Meta data =====
   124→   124→  const [categories, setCategories] = useState<Category[]>([])
   125→   125→  const [channels, setChannels] = useState<Channel[]>([])
   126→   126→
   127→   127→  // ===== Dialog thêm / sửa =====
   128→   128→  const [dialogOpen, setDialogOpen] = useState(false)
   129→   129→  const [editing, setEditing] = useState<Product | null>(null)
   130→   130→  const [form, setForm] = useState<ProductFormState>(EMPTY_FORM)
   131→   131→  const [saving, setSaving] = useState(false)
   132→   132→
   133→   133→  // ===== Xóa sản phẩm =====
   134→   134→  const [deleting, setDeleting] = useState<Product | null>(null)
   135→   135→  const [deleteLoading, setDeleteLoading] = useState(false)
   136→   136→
   137→   137→  // ===== In nhãn mã vạch =====
   138→   138→  const [printOpen, setPrintOpen] = useState(false)
   139→   139→  const [printProducts, setPrintProducts] = useState<BarcodeProduct[]>([])
   140→   140→
   141→   141→  const openPrintDialog = (list: ProductRow[]) => {
   142→   142→    setPrintProducts(
   143→   143→      list.map((p) => ({ id: p.id, sku: p.sku, name: p.name, barcode: p.barcode ?? null, sellPrice: p.sellPrice }))
   144→   144→    )
   145→   145→    setPrintOpen(true)
   146→   146→  }
   147→   147→
   148→   148→  // ===== Tìm kiếm debounce (client-side) =====
   149→   149→  useEffect(() => {
   150→   150→    const timer = setTimeout(() => {
   151→   151→      setSearch(searchInput.trim())
   152→   152→      setPage(1)
   153→   153→    }, 400)
   154→   154→    return () => clearTimeout(timer)
   155→   155→  }, [searchInput])
   156→   156→
   157→   157→  const fetchProducts = useCallback(async () => {
   158→   158→    setLoading(true)
   159→   159→    setError(null)
   160→   160→    try {
   161→   161→      const params = new URLSearchParams()
   162→   162→      if (search) params.set('search', search)
   163→   163→      if (categoryFilter !== 'ALL') params.set('categoryId', categoryFilter)
   164→   164→      if (statusFilter !== 'ALL') params.set('status', statusFilter)
   165→   165→      const qs = params.toString()
   166→   166→      const data = await apiGet<{ products: ProductRow[] }>(`/api/products${qs ? `?${qs}` : ''}`)
   167→   167→      setProducts(data.products)
   168→   168→    } catch (err) {
   169→   169→      setError(errorMessage(err))
   170→   170→    } finally {
   171→   171→      setLoading(false)
   172→   172→    }
   173→   173→  }, [search, categoryFilter, statusFilter])
   174→   174→
   175→   175→  useEffect(() => {
   176→   176→    void fetchProducts()
   177→   177→  }, [fetchProducts])
   178→   178→
   179→   179→  const fetchMeta = useCallback(async () => {
   180→   180→    try {
   181→   181→      const [catRes, chRes] = await Promise.all([
   182→   182→        apiGet<{ categories: Category[] }>('/api/categories'),
   183→   183→        apiGet<{ channels: Channel[] }>('/api/channels'),
   184→   184→      ])
   185→   185→      setCategories(catRes.categories)
   186→   186→      setChannels(chRes.channels)
   187→   187→    } catch (err) {
   188→   188→      toast({
   189→   189→        title: 'Không thể tải danh mục / kênh bán',
   190→   190→        description: errorMessage(err),
   191→   191→        variant: 'destructive',
   192→   192→      })
   193→   193→    }
   194→   194→  }, [toast])
   195→   195→
   196→   196→  useEffect(() => {
   197→   197→    void fetchMeta()
   198→   198→  }, [fetchMeta])
   199→   199→
   200→   200→  // ===== Phân trang client-side =====
   201→   201→  const totalPages = Math.max(1, Math.ceil(products.length / PAGE_SIZE))
   202→   202→  const currentPage = Math.min(page, totalPages)
   203→   203→  const pagedProducts = useMemo(
   204→   204→    () => products.slice((currentPage - 1) * PAGE_SIZE, currentPage * PAGE_SIZE),
   205→   205→    [products, currentPage]
   206→   206→  )
   207→   207→  const hasFilter = search !== '' || categoryFilter !== 'ALL' || statusFilter !== 'ALL'
   208→   208→
   209→   209→  // ===== Dialog handlers =====
   210→   210→  const openCreate = () => {
   211→   211→    setEditing(null)
   212→   212→    setForm(EMPTY_FORM)
   213→   213→    setDialogOpen(true)
   214→   214→  }
   215→   215→
   216→   216→  const openEdit = (p: ProductRow) => {
   217→   217→    setEditing(p)
   218→   218→    setForm({
   219→   219→      sku: p.sku,
   220→   220→      name: p.name,
   221→   221→      description: p.description ?? '',
   222→   222→      categoryId: p.categoryId ?? 'NONE',
   223→   223→      barcode: p.barcode ?? '',
   224→   224→      image: p.image ?? '',
   225→   225→      costPrice: String(p.costPrice),
   226→   226→      sellPrice: String(p.sellPrice),
   227→   227→      stock: String(p.stock),
   228→   228→      lowStockThreshold: String(p.lowStockThreshold),
   229→   229→      status: p.status,
   230→   230→      listingChannelIds: (p.listings ?? []).map((l) => l.channelId),
   231→   231→    })
   232→   232→    setDialogOpen(true)
   233→   233→  }
   234→   234→
   235→   235→  const toggleChannel = (channelId: string, checked: boolean) => {
   236→   236→    setForm((f) => ({
   237→   237→      ...f,
   238→   238→      listingChannelIds: checked
   239→   239→        ? [...f.listingChannelIds, channelId]
   240→   240→        : f.listingChannelIds.filter((id) => id !== channelId),
   241→   241→    }))
   242→   242→  }
   243→   243→
   244→   244→  const handleSave = async () => {
   245→   245→    if (!form.sku.trim()) {
   246→   246→      toast({ title: 'Vui lòng nhập SKU sản phẩm', variant: 'destructive' })
   247→   247→      return
   248→   248→    }
   249→   249→    if (!form.name.trim()) {
   250→   250→      toast({ title: 'Vui lòng nhập tên sản phẩm', variant: 'destructive' })
   251→   251→      return
   252→   252→    }
   253→   253→    if (!Number(form.sellPrice)) {
   254→   254→      toast({ title: 'Vui lòng nhập giá bán hợp lệ', variant: 'destructive' })
   255→   255→      return
   256→   256→    }
   257→   257→    setSaving(true)
   258→   258→    try {
   259→   259→      const base = {
   260→   260→        sku: form.sku.trim(),
   261→   261→        name: form.name.trim(),
   262→   262→        description: form.description.trim(),
   263→   263→        categoryId: form.categoryId === 'NONE' ? null : form.categoryId,
   264→   264→        barcode: form.barcode.trim(),
   265→   265→        image: form.image.trim(),
   266→   266→        costPrice: Number(form.costPrice) || 0,
   267→   267→        sellPrice: Number(form.sellPrice) || 0,
   268→   268→        lowStockThreshold: Number(form.lowStockThreshold) || 0,
   269→   269→        status: form.status,
   270→   270→        listingChannelIds: form.listingChannelIds,
   271→   271→      }
   272→   272→      if (editing) {
   273→   273→        await apiPut<{ product: ProductRow }>(`/api/products/${editing.id}`, base)
   274→   274→        toast({ title: 'Đã cập nhật sản phẩm', description: base.name })
   275→   275→      } else {
   276→   276→        await apiPost<{ product: Product }>('/api/products', { ...base, stock: Number(form.stock) || 0 })
   277→   277→        toast({ title: 'Đã thêm sản phẩm mới', description: base.name })
   278→   278→      }
   279→   279→      setDialogOpen(false)
   280→   280→      void fetchProducts()
   281→   281→    } catch (err) {
   282→   282→      toast({
   283→   283→        title: editing ? 'Không thể cập nhật sản phẩm' : 'Không thể thêm sản phẩm',
   284→   284→        description: errorMessage(err),
   285→   285→        variant: 'destructive',
   286→   286→      })
   287→   287→    } finally {
   288→   288→      setSaving(false)
   289→   289→    }
   290→   290→  }
   291→   291→
   292→   292→  const handleToggleStatus = async (p: Product) => {
   293→   293→    const nextStatus = p.status === 'ACTIVE' ? 'INACTIVE' : 'ACTIVE'
   294→   294→    try {
   295→   295→      await apiPut<{ product: Product }>(`/api/products/${p.id}`, { status: nextStatus })
   296→   296→      toast({
   297→   297→        title: nextStatus === 'ACTIVE' ? 'Đã kích hoạt lại sản phẩm' : 'Đã ngừng bán sản phẩm',
   298→   298→        description: p.name,
   299→   299→      })
   300→   300→      void fetchProducts()
   301→   301→    } catch (err) {
   302→   302→      toast({
   303→   303→        title: 'Cập nhật trạng thái thất bại',
   304→   304→        description: errorMessage(err),
   305→   305→        variant: 'destructive',
   306→   306→      })
   307→   307→    }
   308→   308→  }
   309→   309→
   310→   310→  const handleDelete = async () => {
   311→   311→    if (!deleting) return
   312→   312→    setDeleteLoading(true)
   313→   313→    try {
   314→   314→      await apiDelete<{ success: boolean }>(`/api/products/${deleting.id}`)
   315→   315→      toast({ title: 'Đã xóa sản phẩm', description: deleting.name })
   316→   316→      setDeleting(null)
   317→   317→      void fetchProducts()
   318→   318→    } catch (err) {
   319→   319→      // 409: sản phẩm đã có đơn hàng → gợi ý chuyển sang "Ngừng bán"
   320→   320→      toast({
   321→   321→        title: 'Không thể xóa sản phẩm',
   322→   322→        description: `${errorMessage(err)}. Nếu sản phẩm đã có đơn hàng, hãy chuyển sang "Ngừng bán" thay vì xóa.`,
   323→   323→        variant: 'destructive',
   324→   324→      })
   325→   325→      setDeleting(null)
   326→   326→    } finally {
   327→   327→      setDeleteLoading(false)
   328→   328→    }
   329→   329→  }
   330→   330→
   331→   331→  return (
   332→   332→    <div>
   333→   333→      <PageHeader title="Sản phẩm" description="Quản lý sản phẩm và kênh phân phối đa kênh">
   334→   334→        <Button variant="outline" onClick={() => openPrintDialog(products)}>
   335→   335→          <Barcode className="mr-1.5 h-4 w-4" /> In nhãn mã vạch
   336→   336→        </Button>
   337→   337→        <Button onClick={openCreate}>
   338→   338→          <Plus className="mr-1.5 h-4 w-4" /> Thêm sản phẩm
   339→   339→        </Button>
   340→   340→      </PageHeader>
   341→   341→
   342→   342→      {/* Toolbar */}
   343→   343→      <div className="mb-4 flex flex-col gap-2 sm:flex-row sm:items-center">
   344→   344→        <div className="relative w-full sm:w-72">
   345→   345→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   346→   346→          <Input
   347→   347→            value={searchInput}
   348→   348→            onChange={(e) => setSearchInput(e.target.value)}
   349→   349→            placeholder="Tìm theo tên, SKU hoặc mã vạch..."
   350→   350→            className="pl-8"
   351→   351→            aria-label="Tìm kiếm sản phẩm"
   352→   352→          />
   353→   353→        </div>
   354→   354→        <Select
   355→   355→          value={categoryFilter}
   356→   356→          onValueChange={(v) => {
   357→   357→            setCategoryFilter(v)
   358→   358→            setPage(1)
   359→   359→          }}
   360→   360→        >
   361→   361→          <SelectTrigger className="w-full sm:w-[180px]" aria-label="Lọc theo danh mục">
   362→   362→            <SelectValue placeholder="Tất cả danh mục" />
   363→   363→          </SelectTrigger>
   364→   364→          <SelectContent>
   365→   365→            <SelectItem value="ALL">Tất cả danh mục</SelectItem>
   366→   366→            {categories.map((c) => (
   367→   367→              <SelectItem key={c.id} value={c.id}>
   368→   368→                {c.name}
   369→   369→              </SelectItem>
   370→   370→            ))}
   371→   371→          </SelectContent>
   372→   372→        </Select>
   373→   373→        <Select
   374→   374→          value={statusFilter}
   375→   375→          onValueChange={(v) => {
   376→   376→            setStatusFilter(v)
   377→   377→            setPage(1)
   378→   378→          }}
   379→   379→        >
   380→   380→          <SelectTrigger className="w-full sm:w-[150px]" aria-label="Lọc theo trạng thái">
   381→   381→            <SelectValue placeholder="Tất cả trạng thái" />
   382→   382→          </SelectTrigger>
   383→   383→          <SelectContent>
   384→   384→            <SelectItem value="ALL">Tất cả trạng thái</SelectItem>
   385→   385→            <SelectItem value="ACTIVE">Đang bán</SelectItem>
   386→   386→            <SelectItem value="INACTIVE">Ngừng bán</SelectItem>
   387→   387→          </SelectContent>
   388→   388→        </Select>
   389→   389→      </div>
   390→   390→
   391→   391→      {/* Nội dung chính */}
   392→   392→      {loading ? (
   393→   393→        <TableSkeleton rows={6} />
   394→   394→      ) : error ? (
   395→   395→        <ErrorState message={error} onRetry={() => void fetchProducts()} />
   396→   396→      ) : products.length === 0 ? (
   397→   397→        <EmptyState
   398→   398→          icon={Package}
   399→   399→          title={hasFilter ? 'Không tìm thấy sản phẩm nào' : 'Chưa có sản phẩm nào'}
   400→   400→          description={
   401→   401→            hasFilter
   402→   402→              ? 'Thử thay đổi từ khóa tìm kiếm hoặc bộ lọc.'
   403→   403→              : 'Bắt đầu bằng cách thêm sản phẩm đầu tiên cho cửa hàng.'
   404→   404→          }
   405→   405→          action={
   406→   406→            <Button onClick={openCreate}>
   407→   407→              <Plus className="mr-1.5 h-4 w-4" /> Thêm sản phẩm
   408→   408→            </Button>
   409→   409→          }
   410→   410→        />
   411→   411→      ) : (
   412→   412→        <>
   413→   413→          <div className="overflow-x-auto rounded-lg border">
   414→   414→            <Table className="min-w-[900px]">
   415→   415→              <TableHeader>
   416→   416→                <TableRow>
   417→   417→                  <TableHead>Sản phẩm</TableHead>
   418→   418→                  <TableHead>Danh mục</TableHead>
   419→   419→                  <TableHead>Mã vạch</TableHead>
   420→   420→                  <TableHead>Giá bán</TableHead>
   421→   421→                  <TableHead>Tồn kho</TableHead>
   422→   422→                  <TableHead>Kênh bán</TableHead>
   423→   423→                  <TableHead>Trạng thái</TableHead>
   424→   424→                  <TableHead className="w-12 text-right">
   425→   425→                    <span className="sr-only">Hành động</span>
   426→   426→                  </TableHead>
   427→   427→                </TableRow>
   428→   428→              </TableHeader>
   429→   429→              <TableBody>
   430→   430→                {pagedProducts.map((p) => (
   431→   431→                  <TableRow key={p.id}>
   432→   432→                    <TableCell>
   433→   433→                      <div className="flex items-center gap-3">
   434→   434→                        <ProductAvatar name={p.name} image={p.image} />
   435→   435→                        <div className="min-w-0">
   436→   436→                          <p className="max-w-[220px] truncate font-medium">{p.name}</p>
   437→   437→                          <p className="text-xs text-muted-foreground">{p.sku}</p>
   438→   438→                        </div>
   439→   439→                      </div>
   440→   440→                    </TableCell>
   441→   441→                    <TableCell className="text-sm">{p.category?.name ?? '—'}</TableCell>
   442→   442→                    <TableCell>
   443→   443→                      <div className="flex items-center gap-1.5">
   444→   444→                        {p.barcode ? (
   445→   445→                          <span className="max-w-[140px] truncate font-mono text-xs" title={p.barcode}>
   446→   446→                            {p.barcode}
   447→   447→                          </span>
   448→   448→                        ) : (
   449→   449→                          <span className="text-xs text-muted-foreground">—</span>
   450→   450→                        )}
   451→   451→                        <Button
   452→   452→                          variant="ghost"
   453→   453→                          size="icon"
   454→   454→                          className="h-7 w-7 shrink-0 text-muted-foreground hover:text-foreground"
   455→   455→                          title="In nhãn mã vạch"
   456→   456→                          aria-label={`In nhãn mã vạch cho ${p.sku}`}
   457→   457→                          onClick={() => openPrintDialog([p])}
