   169→// Các trạng thái đích hợp lệ từ trạng thái hiện tại (chỉ tiến theo luồng + có thể hủy)
   170→function allowedNextStatuses(current: string): string[] {
   171→  if (current === 'CANCELLED') return []
   172→  const idx = STATUS_FLOW.indexOf(current as (typeof STATUS_FLOW)[number])
   173→  const forward = idx >= 0 ? STATUS_FLOW.slice(idx + 1) : STATUS_FLOW
   174→  return [...forward, 'CANCELLED']
   175→}

   177→function itemsSummary(order: Order): string {
   178→  if (order.items.length === 0) return '—'
   179→  if (order.items.length === 1) {
   180→    const item = order.items[0]
   181→    return `${item.quantity} × ${item.product?.name ?? 'Sản phẩm'}`
   182→  }
   183→  return `${order.items.length} sản phẩm`
   184→}
